#pragma once

#include <Urho3D/Scene/Component.h>

class Combatant : public Component
{
    URHO3D_OBJECT(Combatant, Component);

public:
    Combatant(Context* context) 
        : Component(context)
        , node_(nullptr)
        , animController_(nullptr)
        , isFrozen_(false)
        , freezeTimer_(0.0f)
        , side_(0)
        , maxHealth_(100)
        , health_(100)
    {}

    virtual void Initialize(Node* node, const ea::string& prefabPath) = 0;
    virtual void Update(float timeStep) = 0;
    virtual bool CheckCollision(Combatant* other) = 0;
    virtual void HandleHit(Combatant* attacker) = 0;

    // Make SetFrozen VIRTUAL so Zombie can override it
    virtual void SetFrozen(bool freeze, bool lockAnimation = true) = 0;

    virtual void Reset() = 0;

    // Common utility accessors
    virtual float GetForwardOffset() const { return 0.0f; } // Added for lunge tracking

    Node* GetNode() const { return node_; }
    Vector3 GetPosition() const { return node_ ? node_->GetWorldPosition() : Vector3::ZERO; }
    void SetPosition(const Vector3& pos) { if (node_) node_->SetWorldPosition(pos); }

    bool IsFrozen() const { return isFrozen_; }

protected:
    Node* node_;
    AnimationController* animController_;
    bool isFrozen_;
    float freezeTimer_;
    int side_;
    int maxHealth_;
    int health_;
};

    fillLight->SetColor(Color(0.4f, 0.4f, 0.44f));
    fillLight->SetBrightness(0.32f);

    cameraNode->SetPosition(Vector3(0.0f, 1.2f, -3.5f)); 
    cameraNode->LookAt(Vector3(0.0f, 0.6f, 1.6f));
    auto* camera = cameraNode->CreateComponent<Camera>();

        camLightNode->SetRotation(cameraNode->GetRotation());

    engine->Exit();

        // Use GetOrCreateComponent. If the controller doesn't exist, this creates it.

        }
    }

}

void Fighter::Initialize(Node* node, const ea::string& idleAnimPath, const ea::string& attackAnimPath)
{
    node_ = node;
    idleAnimPath_ = idleAnimPath;
    attackAnimPath_ = attackAnimPath;

    if (node_)
    {
        // Use GetOrCreateComponent. If the controller doesn't exist, this creates it.
        animController_ = node_->GetOrCreateComponent<AnimationController>();
        
        if (animController_) {
            URHO3D_LOGINFO("Fighter: AnimationController initialized successfully.");
        }
    }

    Reset();
}

void Fighter::Reset()
{
    isSwinging_ = false;
    swingTimer_ = 0.0f;

    if (node_)
    {
        node_->SetPosition(Vector3(0.0f, 0.0f, 0.0f));
        
        RigidBody* body = node_->GetComponent<RigidBody>();
        if (body)
        {
            body->SetLinearVelocity(Vector3::ZERO);
            body->SetAngularVelocity(Vector3::ZERO);
            body->Activate();
        }
    }

    if (animController_)
    {
        // Stop the animation completely when idle
        animController_->Stop(0, 0.2f);
        
        if (!idleAnimPath_.empty())
        {
            animController_->PlayExclusive(idleAnimPath_, 0, true, 0.2f);
            animController_->SetSpeed(idleAnimPath_, 1.0f);
            animController_->SetTime(idleAnimPath_, 0.0f);
        }
    }
}

void Fighter::StartAttack()
{
    if (!animController_) return;

    // 1. Stop all existing animations immediately
    animController_->StopAll(0.0f); 

    // 2. Play the animation
    // 'false' for 'loop' is MANDATORY to stop continuous swinging
    // '0.1f' fade time ensures it snaps to the start
    animController_->PlayExclusive(attackAnimPath_, 0, false, 0.1f);

    // 3. LOCK POSITION: If the animation moves the character, 
    // manually reset the position to origin (or where they should be)
    // Add this line only if you want them to stay in place:
    node_->SetPosition(Urho3D::Vector3::ZERO); 
    
    URHO3D_LOGINFO("Fighter: Animation played. Position forced to Zero.");
}

void Fighter::Update(float timeStep)
{
    if (isSwinging_)
    {
        swingTimer_ += timeStep;
        if (swingTimer_ >= maxSwingDuration_)
        {
            isSwinging_ = false;
            swingTimer_ = 0.0f;
            
            if (animController_ && !idleAnimPath_.empty())
            {
                animController_->PlayExclusive(idleAnimPath_, 0, true, 0.3f);
            }
        }
    }
}

void Zombie::Initialize(Node* node, const ea::string& runAnimPath)
{
    node_ = node;
    runAnimPath_ = runAnimPath;

    if (node_)
    {
        animController_ = node_->GetComponent<AnimationController>();
    }

    isFrozen_ = false;
    isContactFrozen_ = false;
    isNeckBiting_ = false;
    health_ = maxHealth_;

    if (node_)
    {
        Vector3 currentPos = node_->GetPosition();
        node_->SetPosition(Vector3(-4.5f, currentPos.y_, currentPos.z_));
        
        RigidBody* body = node_->GetComponent<RigidBody>();
        if (body)
        {
            body->SetLinearVelocity(Vector3::ZERO);
            body->SetAngularVelocity(Vector3::ZERO);
            body->Activate();
        }
    }

    if (animController_ && !runAnimPath_.empty())
    {
        animController_->PlayExclusive(runAnimPath_, 0, true, 0.0f);
        animController_->SetSpeed(runAnimPath_, 1.0f);
    }
}

void Zombie::Reset()
{
    isFrozen_ = false;
    isContactFrozen_ = false;
    isNeckBiting_ = false;
    health_ = maxHealth_;

    if (node_)
    {
        Vector3 currentPos = node_->GetPosition();
        node_->SetPosition(Vector3(-4.5f, currentPos.y_, currentPos.z_));
        
        RigidBody* body = node_->GetComponent<RigidBody>();
        if (body)
        {
            body->SetLinearVelocity(Vector3::ZERO);
            body->SetAngularVelocity(Vector3::ZERO);
            body->Activate(); 
        }
        
        URHO3D_LOGINFO("[ZOMBIE-RESET] Zombie fully unfrozen and physics body activated.");
    }

    if (animController_ && !runAnimPath_.empty())
    {
        animController_->PlayExclusive(runAnimPath_, 0, true, 0.2f);
        animController_->SetSpeed(runAnimPath_, 1.0f);
        animController_->SetTime(runAnimPath_, 0.0f); 
    }
}

void Zombie::TriggerNeckBite()
{
    if (isNeckBiting_) return;
    
    isNeckBiting_ = true;
    URHO3D_LOGINFO("[ZOMBIE] Starting neck bite sequence!");
    
    RigidBody* body = node_ ? node_->GetComponent<RigidBody>() : nullptr;
    if (body)
    {
        body->SetLinearVelocity(Vector3::ZERO);
    }
}

void Zombie::StartNeckBite(Node* targetNode)
{
    TriggerNeckBite();
}

void Zombie::ApplyBoundaryFreeze()
{
    if (isContactFrozen_) return;

    isContactFrozen_ = true;
    URHO3D_LOGINFO("[ZOMBIE-LOCK] Gate activated. Absolute physics freeze applied at boundary threshold.");

    RigidBody* body = node_ ? node_->GetComponent<RigidBody>() : nullptr;
    if (body)
    {
        body->SetLinearVelocity(Vector3::ZERO);
    }
}

void Zombie::CheckCollision(Fighter* fighter)
{
    if (!node_ || !fighter) return;
    
    float distance = std::abs(node_->GetPosition().x_ - fighter->GetPosition().x_);
    if (distance < 1.2f && !isContactFrozen_)
    {
        ApplyBoundaryFreeze();
    }
}

void Zombie::HandleHit(Fighter* fighter)
{
    URHO3D_LOGINFO("[COMBAT] Zombie registers hit from fighter's weapon.");
    health_ -= 25.0f; 
    if (health_ <= 0.0f)
    {
        isFrozen_ = true;
        URHO3D_LOGINFO("[COMBAT] Zombie defeated!");
    }
}

void Zombie::Update(float timeStep)
{
    if (isFrozen_ || isContactFrozen_ || isNeckBiting_)
    {
        if (node_)
        {
            RigidBody* body = node_->GetComponent<RigidBody>();
            if (body && body->GetLinearVelocity().LengthSquared() > 0.001f)
            {
                body->SetLinearVelocity(Vector3::ZERO);
            }
        }
        return;
    }

    if (node_)
    {
        RigidBody* body = node_->GetComponent<RigidBody>();
        if (body)
        {
            body->SetLinearVelocity(Vector3(runSpeed_, body->GetLinearVelocity().y_, 0.0f));
        }
    }
}

#include "CombatSystem.h"
#include <Urho3D/IO/Log.h>
#include <cmath>

CombatSystem::CombatSystem(Context* context)
    : Component(context)
    , fighter_(nullptr)
    , zombie_(nullptr)
    , debugText_(nullptr)
    , arenaHitFrozen_(false)
{
}

void CombatSystem::Initialize(Fighter* fighter, Zombie* zombie, Text* uiText)
{
    fighter_ = fighter;
    zombie_ = zombie;
    debugText_ = uiText;
    arenaHitFrozen_ = false;
}

void CombatSystem::UpdateCombat(float timeStep)
{
    if (!fighter_ || !zombie_) return;

    // Run collision detection
    if (!arenaHitFrozen_ && !zombie_->IsNeckBiting())
    {
        zombie_->CheckCollision(fighter_);
    }

    // Check if weapon strike connects (NOW CAN INTERRUPT A BITE)
    if (fighter_->IsMidSwing() && !zombie_->IsFrozen() && zombie_->IsContactFrozen() && !arenaHitFrozen_)
    {
        float deltaX = std::abs(fighter_->GetPosition().x_ - zombie_->GetPosition().x_);
        if (deltaX < 1.4f)
        {
            arenaHitFrozen_ = true;
            zombie_->HandleHit(fighter_);

            if (debugText_)
            {
                debugText_->SetText("CHOP! ZOMBIE DEAD! PRESS SPACE TO RESET.");
                debugText_->SetColor(Color::RED);
            }
            return;
        }
    }

    // Bite Sequence with Delay
    if (!arenaHitFrozen_ && zombie_->IsContactFrozen() && !zombie_->IsNeckBiting() && !zombie_->IsFrozen() && !fighter_->IsMidSwing())
    {
        static float biteDelayTimer = 0.0f;
        biteDelayTimer += timeStep;

        if (biteDelayTimer > 0.75f) // Give the player 0.75 seconds to react before biting
        {
            zombie_->StartNeckBite(fighter_->GetNode());
            biteDelayTimer = 0.0f; // Reset timer for the next zombie
            
            if (debugText_)
            {
                debugText_->SetText("NECK BITE! YOU DIED!");
                debugText_->SetColor(Color::YELLOW);
            }
        }
    }
}

void CombatSystem::HandleManualAttack()
{
    if (!fighter_ || !zombie_) return;

    // Reset processing check
    if (zombie_->IsFrozen() || zombie_->IsNeckBiting() || zombie_->IsContactFrozen())
    {
        URHO3D_LOGINFO("[SYSTEM-INPUT] SPACEBAR: Resetting arena stage...");
        
        arenaHitFrozen_ = false;
        fighter_->Reset();
        zombie_->Reset();
        return;
    }

    // Normal Swing Routing
    URHO3D_LOGINFO("[SYSTEM-INPUT] SPACEBAR: Executing fighter axe swing command.");
    fighter_->StartAttack();
}

void CombatSystem::ResetSimulation()
{
    arenaHitFrozen_ = false;
    if (fighter_) fighter_->Reset();
    if (zombie_) zombie_->Reset();
}
