#pragma once

enum SIDE { SIDE_HERO = 0, SIDE_VILLAIN = 1 };

class Combatant : public Component
{
    URHO3D_OBJECT(Combatant, Component);

public:
    Combatant(Context* context) 
        : Component(context)
        , node_(nullptr)
        , animController_(nullptr)
        , isFrozen_(false)
        , freezeTimer_(0.0f)
        , side_(0)
        , maxHealth_(100)
        , health_(100)
    {}

    virtual void Initialize(Node* node, const ea::string& prefabPath) = 0;
    virtual void Update(float timeStep) = 0;
    virtual bool CheckCollision(Combatant* other) = 0;
    virtual void HandleHit(Combatant* attacker) = 0;

    // Make SetFrozen VIRTUAL so Zombie can override it
    virtual void SetFrozen(bool freeze, bool lockAnimation = true) = 0;

    virtual void Reset() = 0;

    // Common utility accessors
    virtual float GetForwardOffset() const { return 0.0f; } // Added for lunge tracking

    Node* GetNode() const { return node_; }
    Vector3 GetPosition() const { return node_ ? node_->GetWorldPosition() : Vector3::ZERO; }
    void SetPosition(const Vector3& pos) { if (node_) node_->SetWorldPosition(pos); }

    bool IsFrozen() const { return isFrozen_; }

protected:
    Node* node_;
    AnimationController* animController_;
    bool isFrozen_;
    float freezeTimer_;
    int side_;
    int maxHealth_;
    int health_;
};

CombatSystem::CombatSystem(Context* context)
    : Component(context)
    , fighter_(nullptr)
    , zombie_(nullptr)
    , debugText_(nullptr)
    , arenaHitFrozen_(false)
{
    // Production hardcoded metrics calibrated from your logs
    zombieOffsetScale_ = 0.820f;
    zombieOffsetRotation_ = 232.0f;
    fighterOffsetRotation_ = -144.0f;
    zombieOffsetPosition_ = Vector3(-0.430f, -0.480f, -0.170f);
    fighterOffsetPosition_ = Vector3(0.000f, -0.110f, 0.000f);
}

void CombatSystem::Initialize(Fighter* fighter, Zombie* zombie, Text* uiText)
{
    fighter_ = fighter;
    zombie_ = zombie;
    debugText_ = uiText;
    arenaHitFrozen_ = false;
}

void CombatSystem::UpdateCombat(float timeStep)
{
    if (!fighter_ || !zombie_) return;

    // 1. Standard collision approach checks
    if (!arenaHitFrozen_ && !zombie_->IsNeckBiting())
    {
        zombie_->CheckCollision(fighter_);
    }

    // 2. Fighter attack strike detection
    if (fighter_->IsMidSwing() && !zombie_->IsFrozen() && zombie_->IsContactFrozen() && !arenaHitFrozen_)
    {
        float deltaX = std::abs(fighter_->GetPosition().x_ - zombie_->GetPosition().x_);
        if (deltaX < 1.4f)
        {
            URHO3D_LOGINFOF("[COMBAT-HIT] Weapon hit! DeltaX gap: %.2f.", deltaX);
            arenaHitFrozen_ = true;
            zombie_->HandleHit(fighter_);
            if (debugText_)
            {
                debugText_->SetText("CHOP DETECTED! PRESS SPACEBAR TO SPAWN NEXT RUNNER.");
                debugText_->SetColor(Color::RED);
            }
            return;
        }
    }

    // 3. Perfect Sync Bite Sequence - Kinematic Asset Swap Transition
    if (!arenaHitFrozen_ && zombie_->IsContactFrozen() && !zombie_->IsNeckBiting() && !zombie_->IsFrozen() && !fighter_->IsMidSwing())
    {
        static float biteDelayTimer = 0.0f;
        biteDelayTimer += timeStep;
        if (biteDelayTimer > 0.75f)
        {
            biteDelayTimer = 0.0f;
            arenaHitFrozen_ = true;
            
            zombie_->StartNeckBite(fighter_->GetNode());
            zombie_->SetBiteSynced(true);
            fighter_->SetBiteSynced(true);
 
            if (debugText_)
            {
                debugText_->SetText("NECK BITE! SOLID KINEMATIC COLLIDERS ACTIVE.");
                debugText_->SetColor(Color::YELLOW);
            }

            Vector3 bitePosition = fighter_->GetPosition();
            Quaternion biteRotation = fighter_->GetRotation();

            // CRITICAL FIX: Hide the meshes but DO NOT disable the deep physics loop container!
            // We turn off the mesh rendering, but let the collision system push back.
            if (fighter_->GetNode()) fighter_->GetNode()->SetDeepEnabled(false);
            if (zombie_->GetNode()) zombie_->GetNode()->SetDeepEnabled(false);

            // Turn on Kinematic parameters so physics engine treats them as solid moving obstacles
            if (fighter_->GetRigidBody()) fighter_->GetRigidBody()->SetKinematic(true);
            if (zombie_->GetRigidBody()) zombie_->GetRigidBody()->SetKinematic(true);

            auto* scene = GetScene();
            auto* cache = GetSubsystem<ResourceCache>();
            Node* biteSceneRoot = scene->CreateChild("SyncedBiteSceneAnchor");
            biteSceneRoot->SetPosition(bitePosition);
            biteSceneRoot->SetRotation(biteRotation);

            // Fighter Instance
            Node* bFighterNode = biteSceneRoot->CreateChild("BiteFighterPrefabNode");
            auto* fPrefab = cache->GetResource<PrefabResource>("fighter-nek-bite/test.glb.d/Prefab.prefab");
            if (fPrefab)
            {
                fPrefab->InstantiateReference(bFighterNode);
                float correctedFighterScale = (5.0f / 9.0f) * 0.01f;
                bFighterNode->SetScale(Vector3(correctedFighterScale, correctedFighterScale, correctedFighterScale));
                bFighterNode->SetPosition(fighterOffsetPosition_);
                bFighterNode->SetRotation(Quaternion(0.0f, fighterOffsetRotation_, 0.0f));

                auto* bFighterAnimCtrl = bFighterNode->FindComponent<AnimationController>();
                if (bFighterAnimCtrl)
                {
                    ea::string fBiteAnim = "fighter-nek-bite/test.glb.d/Animations/Armature.001_Armature.001_mixamo.com_Layer0.ani";
                    AnimationParameters fParams(context_, fBiteAnim);
                    fParams.looped_ = true;
                    bFighterAnimCtrl->PlayNewExclusive(fParams);
                    bFighterAnimCtrl->SetTime(fBiteAnim, 0.0f);
                }
            }

            // Zombie Instance
            Node* bZombieNode = biteSceneRoot->CreateChild("BiteZombiePrefabNode");
            auto* zPrefab = cache->GetResource<PrefabResource>("zombie-nek-bite/test.glb.d/Prefab.prefab");
            if (zPrefab)
            {
                zPrefab->InstantiateReference(bZombieNode);
                float correctedZombieScale = (5.0f / 9.0f) * 0.01f * zombieOffsetScale_;
                bZombieNode->SetScale(Vector3(correctedZombieScale, correctedZombieScale, correctedZombieScale));
                bZombieNode->SetPosition(zombieOffsetPosition_); 
                bZombieNode->SetRotation(Quaternion(0.0f, zombieOffsetRotation_, 0.0f));

                auto* bZombieAnimCtrl = bZombieNode->FindComponent<AnimationController>();
                if (bZombieAnimCtrl)
                {
                    ea::string zBiteAnim = "zombie-nek-bite/test.glb.d/Animations/Armature_Armature_mixamo.com_Layer0_0.ani";
                    AnimationParameters zParams(context_, zBiteAnim);
                    zParams.looped_ = true;
                    bZombieAnimCtrl->PlayNewExclusive(zParams);
                    bZombieAnimCtrl->SetTime(zBiteAnim, 0.0f);
                }
            }
            
            URHO3D_LOGINFO("[SYSTEM-SYNCHRONIZER] Kinematic barriers safely locked to prevent mesh intersection.");
        }
    }
}

void CombatSystem::HandleManualAttack()
{
    if (!fighter_ || !zombie_) return;

    if (zombie_->IsFrozen() || zombie_->IsNeckBiting())
    {
        URHO3D_LOGINFO("[SYSTEM-INPUT] SPACEBAR: Resetting arena stage...");
        ResetSimulation();
        return;
    }

    URHO3D_LOGINFO("[SYSTEM-INPUT] SPACEBAR: Executing fighter axe swing command.");
    fighter_->StartAttack();
}

void CombatSystem::ResetSimulation()
{
    arenaHitFrozen_ = false;

    if (auto* scene = GetScene())
    {
        if (Node* oldBite = scene->GetChild("SyncedBiteSceneAnchor"))
        {
            oldBite->Remove();
        }
    }

    if (fighter_)
    {
        fighter_->GetNode()->SetDeepEnabled(true);
        fighter_->Reset();
    }
    if (zombie_)
    {
        zombie_->GetNode()->SetDeepEnabled(true);
        zombie_->Reset();
    }

    if (debugText_)
    {
        debugText_->SetText("STAGE ACTIVE: RUNNING...");
        debugText_->SetColor(Color::GREEN);
    }
}

#include "Fighter.h"

// Physics collision layers
enum CollisionLayer {
    LAYER_WORLD = 1,
    LAYER_FIGHTER = 2,
    LAYER_ZOMBIE = 4
};

Fighter::Fighter(Context* context) : LogicComponent(context)
{
    SetUpdateEventMask(USE_FIXEDUPDATE);
}

void Fighter::RegisterObject(Context* context)
{
    context->RegisterFactory<Fighter>();
}

void Fighter::Initialize(Node* node, const ea::string& prefabPath)
{
    node_ = node;
    idleAnimPath_.clear();
    attackAnimPath_ = "test.glb.d/Animations/Armature_Armature_mixamo.com_Layer0.ani";

    if (node_)
    {
        arenaPosition_ = node_->GetPosition();
        auto* cache = GetSubsystem<ResourceCache>();
        if (cache && !prefabPath.empty())
        {
            auto* prefabFile = cache->GetResource<PrefabResource>(prefabPath);
            if (prefabFile)
            {
                meshNode_ = node_->CreateChild("AxeManMesh");
                prefabFile->InstantiateReference(meshNode_);
                meshNode_->SetRotation(Quaternion(0.0f, -90.0f, 0.0f));
                meshNode_->SetScale(5.0f / 9.0f);
                
                Vector3 pos = meshNode_->GetPosition();
                pos.y_ = 0.05f;
                meshNode_->SetPosition(pos);

                animController_ = meshNode_->FindComponent<AnimationController>();
                if (animController_) animController_->StopAll();
            }
        }

        if (meshNode_) ReparentWeapon(meshNode_);
        SetupPhysics();
    }
    Reset();
}

void Fighter::Initialize(Node* node, const ea::string& idleAnimPath, const ea::string& attackAnimPath)
{
    node_ = node;
    idleAnimPath_ = idleAnimPath;
    attackAnimPath_ = attackAnimPath;
    if (node_)
    {
        arenaPosition_ = node_->GetPosition();
        animController_ = node_->GetOrCreateComponent<AnimationController>();
        SetupPhysics();
    }
    Reset();
}

void Fighter::SetupPhysics()
{
    if (!node_) return;
    rigidBody_ = node_->GetOrCreateComponent<RigidBody>();
    if (rigidBody_)
    {
        rigidBody_->SetMass(80.0f);
        rigidBody_->SetUseGravity(true); // Gravity active like Ninja Sample
        rigidBody_->SetLinearDamping(10.0f);
        rigidBody_->SetAngularDamping(10.0f);
        rigidBody_->SetKinematic(false);
     rigidBody_->SetCollisionLayer(LAYER_FIGHTER);
     rigidBody_->SetCollisionMask(LAYER_WORLD | LAYER_ZOMBIE); // Collide only with world and zombie

        // NINJA SAMPLE RULE: Freeze all angular rotations (X, Y, Z rotation tipping over)
        // This forces the capsule to stay perfectly vertical on the ground grid at all times
        rigidBody_->SetAngularFactor(Vector3::ZERO); 
    }
    collisionShape_ = node_->GetOrCreateComponent<CollisionShape>();
    if (collisionShape_) collisionShape_->SetCapsule(0.45f, 1.6f, Vector3(0.0f, 0.8f, 0.0f));
}

void Fighter::Reset()
{
    isSwinging_ = false;
    isBiteSynced_ = false;
    swingTimer_ = 0.0f;

    if (node_)
    {
        node_->SetPosition(arenaPosition_);
        node_->SetRotation(Quaternion::IDENTITY);
        if (rigidBody_)
        {
            rigidBody_->SetKinematic(false);
            rigidBody_->SetAngularFactor(Vector3::ZERO); // Re-enforce lock
            rigidBody_->SetPosition(arenaPosition_);
            rigidBody_->SetRotation(Quaternion::IDENTITY);
            rigidBody_->SetLinearVelocity(Vector3::ZERO);
            rigidBody_->ResetForces();
            rigidBody_->Activate();
        }
    }
    ResetMeshLocalTransform();
    if (animController_) animController_->StopAll();
}

void Fighter::ResetMeshLocalTransform()
{
    if (meshNode_)
    {
        meshNode_->SetRotation(Quaternion(0.0f, -90.0f, 0.0f));
        meshNode_->SetScale(5.0f / 9.0f);
        meshNode_->SetPosition(Vector3(0.0f, 0.05f, 0.0f));
    }
}

void Fighter::StartAttack()
{
    if (isSwinging_ || isBiteSynced_) return;
    if (!animController_) return;

    isSwinging_ = true;
    swingTimer_ = 0.0f;
    animController_->StopAll();

    AnimationParameters params(context_, attackAnimPath_);
    params.looped_ = false;
    animController_->PlayNewExclusive(params);
    animController_->SetTime(attackAnimPath_, 0.0f);
    animController_->SetSpeed(attackAnimPath_, 1.0f);
}

void Fighter::FixedUpdate(float timeStep)
{
    if (isBiteSynced_) return;

    if (isSwinging_)
    {
        swingTimer_ += timeStep;
        if (!animController_->IsPlaying(attackAnimPath_))
        {
            isSwinging_ = false;
            swingTimer_ = 0.0f;
            ResetMeshLocalTransform();

            if (rigidBody_)
            {
                rigidBody_->SetPosition(arenaPosition_);
                rigidBody_->SetLinearVelocity(Vector3::ZERO);
                rigidBody_->Activate();
            }
        }
    }

    if (rigidBody_ && !isSwinging_)
    {
        Vector3 pos = rigidBody_->GetPosition();
        Vector3 drift = arenaPosition_ - pos;
        drift.y_ = 0.0f;
        if (drift.Length() > 0.001f)
        {
            rigidBody_->SetPosition(arenaPosition_);
            rigidBody_->SetLinearVelocity(Vector3::ZERO);
        }
        if (node_->GetPosition() != arenaPosition_) node_->SetPosition(arenaPosition_);
    }
}

void Fighter::ReparentWeapon(Node* meshNode)
{
    if (!meshNode) return;
    auto* animatedModel = meshNode->FindComponent<AnimatedModel>();
    if (!animatedModel) return;

    Skeleton& skeleton = animatedModel->GetSkeleton();
    Bone* rightHandBone = skeleton.GetBone("RightHand");
    if (rightHandBone && rightHandBone->node_)
    {
        ea::vector<Node*> allDescendants;
        ea::vector<Node*> stack = { meshNode };
        while (!stack.empty())
        {
            Node* current = stack.back();
            stack.pop_back();
            for (Node* child : current->GetChildren())
            {
                allDescendants.push_back(child);
                stack.push_back(child);
            }
        }
        for (Node* child : allDescendants)
        {
            auto* sm = child->GetComponent<StaticModel>();
            if (!sm) continue;
            Model* mdl = sm->GetModel();
            if (!mdl || std::strstr(mdl->GetName().c_str(), "BattleAxe") == nullptr) continue;

            Vector3 worldPos = child->GetWorldPosition();
            Quaternion worldRot = child->GetWorldRotation();
            child->SetParent(rightHandBone->node_);
            Quaternion handInv = rightHandBone->node_->GetWorldRotation().Inverse();
            child->SetPosition(handInv * (worldPos - rightHandBone->node_->GetWorldPosition()));
            child->SetRotation(handInv * worldRot);
        }
    }
}

class Fighter : public LogicComponent
{
    URHO3D_OBJECT(Fighter, LogicComponent);

public:
    Fighter(Context* context);

    static void RegisterObject(Context* context);

    Quaternion GetRotation() const { return node_ ? node_->GetRotation() : Quaternion::IDENTITY; }

    SharedPtr<Node> meshNode_;
    SharedPtr<AnimationController> animController_;
    SharedPtr<RigidBody> rigidBody_;
    SharedPtr<CollisionShape> collisionShape_;

    ea::string idleAnimPath_;
    ea::string attackAnimPath_;

    bool isBiteSynced_{false};

    // MAIN LOOP
    // ========================================================================
      while (!engine->IsExiting())
      {
          engine->RunFrame();
          float dt = time->GetTimeStep();
  
          // --- ZOMBIE AUTOPSY: run every frame, but only log every 30th to save IO ---
          static int debugFrame = 0;
          debugFrame++;
          bool logNow = (debugFrame % 30 == 1);
  
          if (logNow && zombieComponent)
          {
              Node* zPivot = zombieComponent->GetNode();
              if (zPivot)
              {
                  URHO3D_LOGINFOF("[Z-AUTOPSY] Pivot    pos=%s rot=%s",
                                  zPivot->GetPosition().ToString().c_str(),
                                  zPivot->GetRotation().ToString().c_str());
  
                  Node* zMesh = zPivot->GetChild("ZombieMesh");
                  if (zMesh)
                  {
                      URHO3D_LOGINFOF("[Z-AUTOPSY] Mesh     worldPos=%s scale=%s",
                                      zMesh->GetWorldPosition().ToString().c_str(),
                                      zMesh->GetScale().ToString().c_str());
  
                      AnimatedModel* am = zMesh->FindComponent<AnimatedModel>();
                      if (!am) am = zPivot->FindComponent<AnimatedModel>(); // fallback
                      if (am)
                      {
                          BoundingBox bb = am->GetWorldBoundingBox();
                          URHO3D_LOGINFOF("[Z-AUTOPSY] WorldBB  min=%s max=%s size=%s",
                                          bb.min_.ToString().c_str(),
                                          bb.max_.ToString().c_str(),
                                          bb.Size().ToString().c_str());
                          URHO3D_LOGINFOF("[Z-AUTOPSY] Drawable enabled=%d castShadows=%d",
                                          am->IsEnabled(), am->GetCastShadows());
                      }
                      else
                      {
                          URHO3D_LOGWARNING("[Z-AUTOPSY] NO AnimatedModel found!");
                      }
                  }
                  else
                  {
                      URHO3D_LOGWARNING("[Z-AUTOPSY] NO 'ZombieMesh' child found!");
                  }
  
                  RigidBody* zBody = zombieComponent->GetRigidBody();
                  if (zBody)
                  {
                      URHO3D_LOGINFOF("[Z-AUTOPSY] Physics  pos=%s vel=%s mass=%.1f",
                                      zBody->GetPosition().ToString().c_str(),
                                      zBody->GetLinearVelocity().ToString().c_str(),
                                      zBody->GetMass());
                  }
              }
          }
  
          // --- VISUAL DEBUG DRAW ---
          auto* dbg = scene->GetComponent<DebugRenderer>();
          if (dbg)
          {
              dbg->SetEnabled(true); // make sure it's on

              if (zombieComponent && zombieComponent->GetNode())
              {
                  Node* zPivot = zombieComponent->GetNode();
                  // Red wire sphere at pivot
                  dbg->AddSphere(Sphere(zPivot->GetWorldPosition(), 0.2f), Color::RED);

                  Node* zMesh = zPivot->GetChild("ZombieMesh");
                  if (zMesh)
                  {
                      AnimatedModel* am = zMesh->FindComponent<AnimatedModel>();
                      if (am)
                      {
                          // Green wire bounding box of the actual model
                          dbg->AddBoundingBox(am->GetWorldBoundingBox(), Color::GREEN);
                      }
                  }
              }

              // Also draw fighter for reference
              if (fighterComponent && fighterComponent->GetNode())
              {
                  Node* fPivot = fighterComponent->GetNode();
                  dbg->AddSphere(Sphere(fPivot->GetWorldPosition(), 0.2f), Color::BLUE);
                  AnimatedModel* fam = fPivot->FindComponent<AnimatedModel>();
                  if (!fam && fPivot->GetChild("AxeManMesh"))
                      fam = fPivot->GetChild("AxeManMesh")->FindComponent<AnimatedModel>();
                  if (fam)
                      dbg->AddBoundingBox(fam->GetWorldBoundingBox(), Color::CYAN);
              }
          }

          if (input->GetKeyPress(KEY_SPACE))
         {
             if (combatSystem->IsArenaHitFrozen())
             {
                 combatSystem->ResetSimulation();
             }
             else
             {
                 combatSystem->HandleManualAttack();
             }
         }

         combatSystem->UpdateCombat(dt);

        // Camera controls
        const Vector3 screenCenterAnchor(0.0f, 0.6f, 1.6f);
        Vector3 anchorToCamVector = cameraNode->GetPosition() - screenCenterAnchor;

        bool orbitLeft  = input->GetKeyDown(KEY_A) || input->GetKeyDown(KEY_Q);
        bool orbitRight = input->GetKeyDown(KEY_D) || input->GetKeyDown(KEY_E);

        if (orbitLeft || orbitRight)
        {
            float rotationDirection = orbitLeft ? -90.0f : 90.0f;
            rotationDirection *= dt;
            Quaternion yawOrbitRotation(rotationDirection, Vector3::UP);
            anchorToCamVector = yawOrbitRotation * anchorToCamVector;
            cameraNode->SetPosition(screenCenterAnchor + anchorToCamVector);
        }

        if (input->GetKeyDown(KEY_W)) cameraNode->Translate(Vector3::FORWARD * 3.0f * dt);
        if (input->GetKeyDown(KEY_S)) cameraNode->Translate(Vector3::BACK * 3.0f * dt);

        cameraNode->LookAt(screenCenterAnchor);

        camLightNode->SetPosition(cameraNode->GetPosition());
        camLightNode->SetRotation(cameraNode->GetRotation());

         if (input->GetKeyPress(KEY_ESCAPE)) engine->Exit();
         
     }

    renderer->SetViewport(0, nullptr);
    engine->Exit();

    return 0;
}

Zombie::Zombie(Context* context) : LogicComponent(context)

    SetUpdateEventMask(USE_FIXEDUPDATE);

void Zombie::RegisterObject(Context* context)
{
    context->RegisterFactory<Zombie>();
}

void Zombie::Initialize(Node* node, const ea::string& prefabPath)
{
    node_ = node;
    if (!node_) return;

    startPosition_ = node_->GetPosition();
    node_->SetRotation(Quaternion(0.0f, 90.0f, 0.0f));

    size_t lastSlash = prefabPath.find_last_of('/');
    runAnimPath_ = (lastSlash != ea::string::npos) ? prefabPath.substr(0, lastSlash) + "/Animations/mixamo.com.ani" : prefabPath;

    meshNode_ = node_->CreateChild("ZombieMesh");
    auto* cache = GetSubsystem<ResourceCache>();
    auto* prefab = cache->GetResource<PrefabResource>(prefabPath);
    if (prefab)
    {
        prefab->InstantiateReference(meshNode_);
        animController_ = meshNode_->FindComponent<AnimationController>();
        meshNode_->SetScale(75.0f);
        meshNode_->SetPosition(Vector3::ZERO);
        meshNode_->SetRotation(Quaternion::IDENTITY);
        SetupPhysics();
    }
    Reset();
}

void Zombie::SetupPhysics()
{
    if (!node_) return;
    rigidBody_ = node_->GetOrCreateComponent<RigidBody>();
    if (rigidBody_)
    {
        rigidBody_->SetMass(60.0f);
        rigidBody_->SetUseGravity(false);
        rigidBody_->SetLinearDamping(5.0f);
        rigidBody_->SetAngularDamping(10.0f);
        rigidBody_->SetKinematic(false); // Default active movement
        rigidBody_->SetCollisionLayer(4);
        rigidBody_->SetCollisionMask(0xFFFFFFFF);
    }
    collisionShape_ = node_->GetOrCreateComponent<CollisionShape>();
    if (collisionShape_) collisionShape_->SetCapsule(0.45f, 1.6f, Vector3(0.0f, 0.8f, 0.0f)); // Match fighter capsule thickness
}

void Zombie::FixedUpdate(float timeStep)
{
    if (isBiteSynced_ || health_ <= 0 || isFrozen_) return;

    if (!isContactFrozen_)
    {
        Vector3 pos = node_->GetPosition();
        pos.x_ += runSpeed_ * timeStep;
        node_->SetPosition(pos);

        if (rigidBody_)
        {
            rigidBody_->SetPosition(pos);
            rigidBody_->SetLinearVelocity(Vector3::ZERO);
        }
    }
    else if (rigidBody_)
    {
        rigidBody_->SetLinearVelocity(Vector3::ZERO);
    }
}

void Zombie::CheckCollision(Fighter* fighter)
{
    if (!node_ || !fighter || isFrozen_ || health_ <= 0 || isNeckBiting_ || isBiteSynced_) return;

    Vector3 myPos = node_->GetPosition();
    Vector3 targetPos = fighter->GetPosition();
    float boundaryX = targetPos.x_ - solidStopThreshold_;
    float deltaX = myPos.x_ - targetPos.x_;

    if (isContactFrozen_)
    {
        if (myPos.x_ > boundaryX)
        {
            node_->SetPosition(Vector3(boundaryX, myPos.y_, myPos.z_));
            if (rigidBody_)
            {
                rigidBody_->SetPosition(Vector3(boundaryX, myPos.y_, myPos.z_));
                rigidBody_->SetLinearVelocity(Vector3::ZERO);
            }
        }
        return;
    }

    if (myPos.x_ >= boundaryX)
    {
        node_->SetPosition(Vector3(boundaryX, myPos.y_, myPos.z_));
        if (rigidBody_)
        {
            rigidBody_->SetPosition(Vector3(boundaryX, myPos.y_, myPos.z_));
            rigidBody_->SetLinearVelocity(Vector3::ZERO);
        }
        // FIXED: Appended the correct member underscore variable suffix here
        isContactFrozen_ = true;
    }
}

void Zombie::HandleHit(Fighter* fighter)
{
    if (hasBeenHit_) return;
    hasBeenHit_ = true;
    isFrozen_ = true;
    if (animController_) animController_->StopAll();
    if (rigidBody_)
    {
        rigidBody_->SetLinearVelocity(Vector3::ZERO);
        rigidBody_->SetAngularVelocity(Vector3::ZERO);
    }
}

void Zombie::ApplyBoundaryFreeze()
{
    if (isContactFrozen_) return;
    isContactFrozen_ = true;
    if (rigidBody_) rigidBody_->SetLinearVelocity(Vector3::ZERO);
}

void Zombie::TriggerNeckBite()
{
    if (isNeckBiting_) return;
    isNeckBiting_ = true;
    if (rigidBody_) rigidBody_->SetLinearVelocity(Vector3::ZERO);
}

void Zombie::StartNeckBite(Node* targetNode)
{
    TriggerNeckBite();
}

void Zombie::Reset()
{
    isFrozen_ = false;
    isContactFrozen_ = false;
    isNeckBiting_ = false;
    isBiteSynced_ = false;
    hasBeenHit_ = false;
    health_ = maxHealth_;

    if (node_)
    {
        node_->SetPosition(startPosition_);
        node_->SetRotation(Quaternion(0.0f, 90.0f, 0.0f));
        if (rigidBody_)
        {
            rigidBody_->SetKinematic(false); // Reset to standard dynamics
            rigidBody_->SetPosition(startPosition_);
            node_->SetRotation(Quaternion(0.0f, 90.0f, 0.0f));
            rigidBody_->SetLinearVelocity(Vector3::ZERO);
            rigidBody_->ResetForces();
            rigidBody_->Activate();
        }
    }

    if (animController_ && !runAnimPath_.empty())
    {
        animController_->StopAll();
        AnimationParameters params(context_, runAnimPath_);
        params.looped_ = true;
        animController_->PlayNewExclusive(params);
        animController_->SetTime(runAnimPath_, 0.0f);
        animController_->SetSpeed(runAnimPath_, 0.5f);
    }
}
